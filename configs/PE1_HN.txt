! --- 1. ĐỊNH NGHĨA VRF VÀ RD/RT CHO TỪNG TENANT ---
ip vrf VRF_CORP_A
 rd 65000:100
 route-target export 65000:100
 route-target import 65000:100
!
ip vrf VRF_CORP_B
 rd 65000:200
 route-target export 65000:200
 route-target import 65000:200

! --- 2. GÁN VRF VÀO SUB-INTERFACE KẾT NỐI VỚI CE1 ---
interface Ethernet0/1.10
 encapsulation dot1Q 10
 ip vrf forwarding VRF_CORP_A
 ip address 10.10.1.1 255.255.255.252
!
interface Ethernet0/1.20
 encapsulation dot1Q 20
 ip vrf forwarding VRF_CORP_B
 ip address 10.10.20.1 255.255.255.252

! --- 3. CỔNG CORE KẾT NỐI P1_CORE ---
interface Ethernet0/0
 ip address 172.16.12.1 255.255.255.252
 mpls ip
 mpls ldp discovery transport-address interface

! --- 4. BẬT BẢN TIN ICMP TTL CHO TRACEROUTE MPLS ---
mpls ip propagate-ttl

! --- 5. BGP VPNv4 VÀ eBGP PE-CE ---
router bgp 65000
 bgp router-id 10.0.0.1
 neighbor 10.0.0.3 remote-as 65000
 neighbor 10.0.0.3 update-source Loopback0
 !
 address-family vpnv4
  neighbor 10.0.0.3 activate
  neighbor 10.0.0.3 send-community extended
 exit-address-family
 !
 address-family ipv4 vrf VRF_CORP_A
  neighbor 10.10.1.2 remote-as 65001
  neighbor 10.10.1.2 activate
 exit-address-family
 !
 address-family ipv4 vrf VRF_CORP_B
  neighbor 10.10.20.2 remote-as 65001
  neighbor 10.10.20.2 activate
 exit-address-family

! --- 6. OSPF CORE VÀ LDP SYNCHRONIZATION ---
router ospf 1
 router-id 10.0.0.1
 network 10.0.0.1 0.0.0.0 area 0
 network 172.16.12.0 0.0.0.3 area 0
 mpls ldp sync
 mpls ldp autoconfig area 0

! --- 7. LDP HARDENING ---
mpls ldp session protection

! --- 1. KẾT NỐI VÀ LDP TRÊN CÁC CỔNG TRANSIT ---
interface Ethernet0/0
 ip address 172.16.12.2 255.255.255.252
 mpls ip
 mpls ldp discovery transport-address interface
!
interface Ethernet0/1
 ip address 172.16.23.1 255.255.255.252
 mpls ip
!
interface Ethernet0/2
 ip address 172.16.24.1 255.255.255.252
 mpls ip

! --- 2. BẬT BẢN TIN ICMP TTL CHO PHÉP TRACEROUTE QUA CORE ---
mpls ip ttl-expiration pop 1

! --- 3. OSPF VÀ LDP SYNCHRONIZATION ---
router ospf 1
 router-id 10.0.0.2
 network 10.0.0.2 0.0.0.0 area 0
 network 172.16.12.0 0.0.0.3 area 0
 network 172.16.23.0 0.0.0.3 area 0
 network 172.16.24.0 0.0.0.3 area 0
 mpls ldp sync
 mpls ldp autoconfig area 0

! --- 4. LDP SESSION PROTECTION ---
mpls ldp session protection

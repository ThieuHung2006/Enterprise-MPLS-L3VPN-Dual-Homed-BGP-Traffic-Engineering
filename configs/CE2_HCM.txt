! --- 1. CẤU HÌNH VRF-LITE CHO 2 TENANT ---
ip vrf CORP_A
 rd 65002:10
!
ip vrf CORP_B
 rd 65002:20

! --- 2. GÁN VRF VÀO CÁC SUB-INTERFACE VÀ LOOPBACK LAN ---
interface Loopback10
 ip vrf forwarding CORP_A
 ip address 192.168.20.1 255.255.255.0
!
interface Loopback100
 ip vrf forwarding CORP_B
 ip address 192.168.200.1 255.255.255.0
!
interface Ethernet0/0.10
 encapsulation dot1Q 10
 ip vrf forwarding CORP_A
 ip address 10.10.2.2 255.255.255.252
!
interface Ethernet0/0.20
 encapsulation dot1Q 20
 ip vrf forwarding CORP_B
 ip address 10.10.30.2 255.255.255.252

! --- 3. BGP eBGP VỚI PE2 THEO TỪNG VRF ---
router bgp 65002
 bgp router-id 192.168.20.1
 !
 address-family ipv4 vrf CORP_A
  network 192.168.20.0
  neighbor 10.10.2.1 remote-as 65000
  neighbor 10.10.2.1 activate
 exit-address-family
 !
 address-family ipv4 vrf CORP_B
  network 192.168.200.0
  neighbor 10.10.30.1 remote-as 65000
  neighbor 10.10.30.1 activate
 exit-address-family

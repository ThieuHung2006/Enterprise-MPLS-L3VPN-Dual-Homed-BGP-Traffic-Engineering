hostname ASBR1_Core
!
ip routing
!
interface Loopback0
 ip address 10.0.0.4 255.255.255.255
 ip ospf 1 area 0
!
interface Ethernet0/0
 description TO_P1_CORE
 ip address 172.16.24.4 255.255.255.0
 ip ospf 1 area 0
 bfd interval 150 min_rx 150 multiplier 3
 no shutdown
!
interface Ethernet0/1
 description TO_ISP1_AS100
 ip address 203.0.113.1 255.255.255.252
 bfd interval 150 min_rx 150 multiplier 3
 no shutdown
!
interface Ethernet0/2
 description TO_ISP2_AS200
 ip address 203.0.113.5 255.255.255.252
 bfd interval 150 min_rx 150 multiplier 3
 no shutdown
!
router ospf 1
 router-id 10.0.0.4
 bfd all-interfaces
 passive-interface Ethernet0/1
 passive-interface Ethernet0/2
!
router bgp 65000
 bgp log-neighbor-changes
 network 203.0.113.128 mask 255.255.255.128
 ! eBGP to ISP1
 neighbor 203.0.113.2 remote-as 100
 neighbor 203.0.113.2 fall-over bfd
 neighbor 203.0.113.2 route-map RM_FROM_ISP1_IN in
 neighbor 203.0.113.2 route-map RM_TO_ISP1_OUT out
 ! eBGP to ISP2
 neighbor 203.0.113.6 remote-as 200
 neighbor 203.0.113.6 fall-over bfd
 neighbor 203.0.113.6 route-map RM_TO_ISP2_OUT out
!
! Prefix-list chi cho phep dai IP so huu cua AS 65000 (chan Transit Leak)
ip prefix-list PL_OWN_PUBLIC seq 5 permit 203.0.113.128/25
!
! Chinh sach dinh tuyen Inbound / Outbound
route-map RM_FROM_ISP1_IN permit 10
 set local-preference 200
!
route-map RM_TO_ISP1_OUT permit 10
 match ip address prefix-list PL_OWN_PUBLIC
!
route-map RM_TO_ISP2_OUT permit 10
 match ip address prefix-list PL_OWN_PUBLIC
 set as-path prepend 65000 65000 65000
!
end
